\begin{problem}
Determine the \emph{reactions} at the supports $A$ and $B$ for the beam loaded as shown
in the following figure.

\begin{center}
\begin{tikzpicture}[x=\u cm,y=1cm,>=Stealth]
  % load diagram
  \fill[loadfill] (0,0.12) -- (3,1.3) -- (8,1.3) -- (12,0.12) -- cycle;
  \draw[loadline,thick] (0,0.12) -- (3,1.3) -- (8,1.3) -- (12,0.12);
  \foreach \x in {0.5,1,...,11.5}{
    \pgfmathsetmacro{\h}{\x<3 ? 0.12+1.18*\x/3 : (\x<=8 ? 1.3 : 0.12+1.18*(12-\x)/4)}
    \draw[->,loadline] (\x,\h) -- (\x,0.14);}
  \node[above,loadline] at (5.5,1.3) {$w_0 = 160\ \mathrm{N/m}$};
  % beam
  \fill[white] (0,-0.0) rectangle (12,0.12);
  \draw[thick] (0,0) rectangle (12,0.12);
  \pinsupport{3}\rollersupport{8}
  \node[above left] at (3,0.12) {\small$A$};
  \node[above right] at (8,0.12) {\small$B$};
  % dimensions
  \foreach \a/\b/\t in {0/3/3\m,3/8/5\m,8/12/4\m}{
    \draw[<->,teal] (\a,-0.85) -- (\b,-0.85) node[midway,below,text=black]{\small$\t$};}
  \foreach \x in {0,3,8,12} \draw[teal!60,dashed] (\x,-0.55) -- (\x,-1.0);
\end{tikzpicture}
\end{center}
\end{problem}

\begin{given}
\begin{itemize}[leftmargin=1.3em,itemsep=0pt,topsep=0pt]
  \item Beam length $12\m$: left overhang $3\m$, span $AB=5\m$, right overhang $4\m$.
  \item $A$ is a \textbf{pin} (reactions $A_x$, $A_y$); $B$ is a \textbf{roller} (reaction $B_y$).
  \item Load rises linearly from $0$ to $160\ \mathrm{N/m}$ over the left overhang,
        is uniform $160\ \mathrm{N/m}$ between $A$ and $B$, and falls linearly to $0$
        over the right overhang.
\end{itemize}
\end{given}

\section{Replace the distributed load by resultant forces}

The trapezoid is split into two triangles and one rectangle. The resultant of each part
equals its \textbf{area} and acts through its \textbf{centroid}. For a triangle, the
centroid lies $\tfrac13$ of its length from the tall side (the support).

\begin{tabular}{>{\columncolor{softteal}\bfseries}l c c}
\toprule
\rowcolor{teal}\color{white}Part & \color{white}\textbf{Resultant} &
  \color{white}\textbf{Position measured from $A$}\\
\midrule
Left triangle   & $W_1=\tfrac12(3)(160)=240\N$ & $\tfrac13(3)=1\m$ \textbf{left} of $A$\\
Rectangle       & $W_2=(5)(160)=800\N$         & $2.5\m$ right of $A$\\
Right triangle  & $W_3=\tfrac12(4)(160)=320\N$ & $5+\tfrac13(4)=6.333\m$ right of $A$\\
\midrule

\bottomrule
\end{tabular}

\section{Equilibrium equations}

\textbf{(a) Horizontal forces.} There is no horizontal load, so
\[
  \sum F_x = 0 \;\Longrightarrow\; A_x = 0 .
\]

\textbf{(b) Moments about $A$} (counter-clockwise positive). $W_1$ lies to the left of
$A$, so its moment is counter-clockwise:
\[
  \sum M_A = 0:\qquad R_B(5) - 800(2.5) - 320(6.333) + 240(1) = 0
\]
\[
  5R_B = 2000 + 2026.67 - 240 = 3786.67
  \quad\Longrightarrow\quad
  {\color{crimson}\boxed{R_B = \tfrac{2272}{3} \approx 757.3\N\ \uparrow}}
\]

\textbf{(c) Vertical forces.}
\[
  \sum F_y = 0:\qquad R_A + R_B - 240 - 800 - 320 = 0
  \quad\Longrightarrow\quad
  {\color{crimson}\boxed{R_A = \tfrac{1808}{3} \approx 602.7\N\ \uparrow}}
\]

\begin{checkbox}
Take moments about $B$ instead (counter-clockwise positive):
\[
  -R_A(5) + 240(6) + 800(2.5) - 320(1.333) = 0
\]
\[
  5R_A = 1440 + 2000 - 426.7 = 3013.3
  \;\Longrightarrow\; R_A = 602.7\N \ \checkmark
\]
\end{checkbox}

\begin{note}
A frequently quoted answer, $R_A=624\N$ and $R_B=736\N$, also adds up to $1360\N$ but
does \textbf{not} satisfy $\sum M_A=0$:
$\;5(736)-3786.67=-106.7\ \mathrm{N\cdot m}\neq 0$.
It comes from placing $W_3$ at $1\m$ from $B$ instead of the correct $\tfrac43\m$.
\end{note}

\begin{answer}
\[
  A_x = 0, \qquad
  R_A = \frac{1808}{3}\approx 602.7\N\ \uparrow, \qquad
  R_B = \frac{2272}{3}\approx 757.3\N\ \uparrow
\]
\end{answer}
